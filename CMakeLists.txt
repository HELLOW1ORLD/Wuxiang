cmake_minimum_required(VERSION 3.25)
project(Wuxiang VERSION 0.1.1 LANGUAGES C CXX)
find_package(LLVM 20.1.8 EXACT REQUIRED CONFIG)
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} from ${LLVM_DIR}")
# The official Windows 20.1.8 SDK exports its builder's absolute DIA path.
# Resolve that one dependency locally without rewriting the vendor package.
if(WIN32 AND TARGET LLVMDebugInfoPDB)
  get_target_property(WUXIANG_PDB_LINKS LLVMDebugInfoPDB INTERFACE_LINK_LIBRARIES)
  foreach(WUXIANG_PDB_LINK IN LISTS WUXIANG_PDB_LINKS)
    if(WUXIANG_PDB_LINK MATCHES "diaguids\\.lib$" AND NOT EXISTS "${WUXIANG_PDB_LINK}")
      find_library(WUXIANG_DIAGUIDS_LIBRARY NAMES diaguids
        HINTS "${CMAKE_GENERATOR_INSTANCE}/DIA SDK/lib/amd64" "$ENV{VSINSTALLDIR}/DIA SDK/lib/amd64")
      if(NOT WUXIANG_DIAGUIDS_LIBRARY)
        message(FATAL_ERROR "Set WUXIANG_DIAGUIDS_LIBRARY to the local x64 DIA SDK diaguids.lib")
      endif()
      list(REMOVE_ITEM WUXIANG_PDB_LINKS "${WUXIANG_PDB_LINK}")
      list(APPEND WUXIANG_PDB_LINKS "${WUXIANG_DIAGUIDS_LIBRARY}")
    endif()
  endforeach()
  set_target_properties(LLVMDebugInfoPDB PROPERTIES INTERFACE_LINK_LIBRARIES "${WUXIANG_PDB_LINKS}")
endif()
# A second exported SDK string joins two linker arguments with a space.
if(WIN32 AND TARGET LLVMSupport)
  get_target_property(WUXIANG_SUPPORT_LINKS LLVMSupport INTERFACE_LINK_LIBRARIES)
  string(REPLACE "> -INCLUDE:malloc" ">;-INCLUDE:malloc" WUXIANG_SUPPORT_LINKS "${WUXIANG_SUPPORT_LINKS}")
  set_target_properties(LLVMSupport PROPERTIES INTERFACE_LINK_LIBRARIES "${WUXIANG_SUPPORT_LINKS}")
endif()
add_executable(wuxiang-ir src/transform.cpp)
target_compile_features(wuxiang-ir PRIVATE cxx_std_17)
target_include_directories(wuxiang-ir SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(WUXIANG_LLVM_DEFINITIONS NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_options(wuxiang-ir PRIVATE ${WUXIANG_LLVM_DEFINITIONS})
llvm_map_components_to_libnames(WUXIANG_LLVM_LIBS Core IRReader BitReader BitWriter Analysis Support)
target_link_libraries(wuxiang-ir PRIVATE ${WUXIANG_LLVM_LIBS})
set_target_properties(wuxiang-ir PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_compile_options(wuxiang-ir PRIVATE /W4 /EHsc /permissive-)
endif()
install(TARGETS wuxiang-ir RUNTIME DESTINATION bin)
install(FILES include/wuxiang/protect.h DESTINATION include/wuxiang)
install(FILES cmake/Wuxiang.cmake DESTINATION lib/cmake/Wuxiang)
